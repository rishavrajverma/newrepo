       <script>
      document.addEventListener(\"DOMContentLoaded\", function () {
        const jsonData ="&$$jsondata&";

        let currentIndex = 0;
        let slideWidth = 0; 
        const gaugesPerSlide = 6; 
        function moveSlider(direction) {
          const container = document.querySelector(\".slider\");
          const slides = container.querySelectorAll(\".slide\");
          const totalSlides = slides.length;

          if (direction === \"left\" && currentIndex > 0) {
            currentIndex--;
          } else if (direction === \"right\" && currentIndex < totalSlides - 1) {
            currentIndex++;
          }

          const offset = -currentIndex * slideWidth;
          container.style.transform = `translateX(${offset}px)`;
        }

        function populateGauges(gaugesData) {
          const container = document.getElementById(\"compliance-tracking\");
          const totalGauges = gaugesData.length;
          const totalSlides = Math.ceil(totalGauges / gaugesPerSlide);

          for (let i = 0; i < totalSlides; i++) {
            const slide = document.createElement(\"div\");
            slide.className = \"slide\";

            for (
              let j = i * gaugesPerSlide;
              j < (i + 1) * gaugesPerSlide && j < totalGauges;
              j++
            ) {
              const gaugeData = gaugesData[j];
              const gauge = createGauge(gaugeData);
              slide.appendChild(gauge);
            }

            container.appendChild(slide);
          }

          
          slideWidth = container.querySelector(\".slide\").offsetWidth;
        }

        function getColor(value) {
          if (value >= 75) return \"#f44336\"; 
          if (value >= 50) return \"#ffeb3b\";
          return \"#4caf50\"; 
        }

        function createGauge(gaugeData) {
          const gauge = document.createElement(\"div\");
          gauge.className = \"gauge\";

          const svgNamespace = \"http://www.w3.org/2000/svg\";
          const svg = document.createElementNS(svgNamespace, \"svg\");
          svg.setAttribute(\"viewBox\", \"0 0 100 100\");

          const circle = document.createElementNS(svgNamespace, \"circle\");
          circle.setAttribute(\"cx\", \"50\");
          circle.setAttribute(\"cy\", \"50\");
          circle.setAttribute(\"r\", \"40\");
          circle.setAttribute(\"fill\", \"none\");
          circle.setAttribute(\"stroke\", \"#f0f0f0\");
          circle.setAttribute(\"stroke-width\", \"10\");

          const path = document.createElementNS(svgNamespace, \"path\");
          const percentage = gaugeData.value === 100 ? 0.99 : gaugeData.value / 100;
          const startX = 50 + 40 * Math.cos(-Math.PI / 2);
          const startY = 50 + 40 * Math.sin(-Math.PI / 2);
          const endX =
            50 + 40 * Math.cos(-Math.PI / 2 + 2 * Math.PI * percentage);
          const endY =
            50 + 40 * Math.sin(-Math.PI / 2 + 2 * Math.PI * percentage);
          const largeArcFlag = percentage > 0.5 ? 1 : 0;
          const pathData = `M ${startX} ${startY} A 40 40 0 ${largeArcFlag} 1 ${endX} ${endY}`;
          path.setAttribute(\"d\", pathData);
          path.setAttribute(\"fill\", \"none\");
          path.setAttribute(\"stroke\", getColor(gaugeData.value));
          path.setAttribute(\"stroke-width\", \"9\");
          

          const textGroup = document.createElementNS(svgNamespace, \"g\");
          textGroup.setAttribute(\"transform\", \"translate(50, 50)\");

          const textValue = document.createElementNS(\"http://www.w3.org/2000/svg\", \"text\");





const valueStr = gaugeData.value.toString();
const valueLength = valueStr.length;

let dxOffset = 0;
if (valueLength <= 2) dxOffset = 8;
else if (valueLength === 3) dxOffset = 16;
else dxOffset = 20;

textValue.setAttribute(\"x\", \"0\");
textValue.setAttribute(\"y\", \"8\");
textValue.setAttribute(\"text-anchor\", \"middle\");
textValue.setAttribute(\"class\", \"gauge-value-text\");
textValue.textContent = valueStr;

const textPercentage = document.createElementNS(\"http://www.w3.org/2000/svg\", \"text\");
textPercentage.setAttribute(\"x\", \"0\");
textPercentage.setAttribute(\"y\", \"8\");
textPercentage.setAttribute(\"dx\", dxOffset.toString());
textPercentage.setAttribute(\"text-anchor\", \"start\");
textPercentage.setAttribute(\"class\", \"gauge-percent-text\");
textPercentage.textContent = \"%\";

textGroup.appendChild(textValue);
textGroup.appendChild(textPercentage);





          

          svg.appendChild(circle);
          svg.appendChild(path);
          svg.appendChild(textGroup);

          gauge.appendChild(svg);

          const title = document.createElement(\"div\");
          title.className = \"gauge-title\";
          title.textContent = gaugeData.title;
          gauge.appendChild(title);

          const gaugeInfo = document.createElement(\"div\");
          gaugeInfo.className = \"gauge-info\";
          gaugeInfo.innerHTML = `<p style=\"line-height: 0.5;\">(${gaugeData.issues} Issues)</p>`;
          gauge.appendChild(gaugeInfo);

          return gauge;
        }
