      rel=\"stylesheet"\ />
    <style>
      body {
        font-family: Arial, sans-serif;
        padding: 20px;
      }

      .table-wrapper {
        max-height: 200px; /* Adjust height to show only 3 rows approx */
        overflow-y: auto;
      }

      table {
        width: 100%;
        border-collapse: collapse;
      }

      th,
      td {
        border: 1px solid #ccc;
        padding: 8px 10px;
        text-align: center;
        white-space: nowrap;
      }

      th {
        background-color: #f5f5f5;
        position: sticky;
        top: 0;
        z-index: 1;
      }

      tr:nth-child(even) {
        background-color: #fafafa;
      }
    </style>
  </head>
  <body>
    <h1 class=\"mb-4\">Grouped Data by Country</h1>
    <div class=\"accordion\" id="accordionData"></div>

    <script>
      const jsonData ="&$$tableJSON&";

      const groupedByCountry = {};

      for (const station in jsonData) {
        const entry = jsonData[station];
        const country = entry.country;

        if (!groupedByCountry[country]) {
          groupedByCountry[country] = [];
        }

        groupedByCountry[country].push({ code: station, ...entry });
      }

      const accordion = document.getElementById(\"accordionData\");
      let index = 0;

      for (const country in groupedByCountry) {
        const countryId = `country-${index}`;
        const collapseId = `collapse-${index}`;
        const data = groupedByCountry[country];

        const accordionItem = document.createElement(\"div\");
        accordionItem.className = \"accordion-item\";

        accordionItem.innerHTML = `
  <h2 class="accordion-header" id="${countryId}">
    <button class="accordion-button ${
      index !== 0 ? "collapsed" : ""
    }" type="button" data-bs-toggle="collapse" data-bs-target="#${collapseId}" aria-expanded="${
          index === 0
        }" aria-controls="${collapseId}">
      ${country}
    </button>
  </h2>
  <div id="${collapseId}" class="accordion-collapse collapse ${
          index === 0 ? "show" : ""
        }" aria-labelledby="${countryId}" data-bs-parent="#accordionData">
    <div class="accordion-body">
      <div class="table-wrapper">
        <table data-country="${country}" class=\"sortable-table\">
          <thead>
            <tr>
              <th>Station Code</th>
              <th class="sortable" data-key=\"city\">City &#x25B2;&#x25BC;</th>
              <th class="sortable" data-key="country">Country &#x25B2;&#x25BC;</th>
              <th>Compliance</th>
              <th>Delays</th>
              <th>Provisioning</th>
              <th>Quality</th>
              <th>Safety</th>
              <th>Unclassified</th>
              <th>Closed</th>
              <th>Unresolved</th>
              <th>Total</th>
            </tr>
          </thead>
          <tbody>
            ${data
              .map(
                (row) => `
              <tr>
                <td>${row.code}</td>
                <td>${row.city}</td>
                <td>${row.country}</td>
                <td>${row.Compliance}</td>
                <td>${row.Delays}</td>
                <td>${row.Provisioning}</td>
                <td>${row.Quality}</td>
                <td>${row.Safety}</td>
                <td>${row.Unclassified}</td>
                <td>${row.closedCount}</td>
                <td>${row.unresolvedCount}</td>
                <td>${row.totalCount}</td>
              </tr>
            `
              )
              .join("")}
          </tbody>
        </table>
      </div>
    </div>
  </div>
`;

        accordion.appendChild(accordionItem);
        index++;
      }

      document.addEventListener("click", function (e) {
        if (!e.target.classList.contains(\"sortable\")) return;

        const th = e.target;
        const key = th.dataset.key;
        const table = th.closest(\"table\");
        const tbody = table.querySelector(\"tbody\");
        const rows = Array.from(tbody.querySelectorAll(\"tr\"));

        const ascending = th.dataset.order !== \"asc\";
        th.dataset.order = ascending ? "asc" : \"desc\";

        rows.sort((a, b) => {
          const aText = a
            .querySelector(`td:nth-child(${th.cellIndex + 1})`)
            .textContent.trim()
            .toLowerCase();
          const bText = b
            .querySelector(`td:nth-child(${th.cellIndex + 1})`)
            .textContent.trim()
            .toLowerCase();
          return ascending
            ? aText.localeCompare(bText)
            : bText.localeCompare(aText);
        });

        tbody.innerHTML = "";
        rows.forEach((row) => tbody.appendChild(row));
      });
    </script>
